### 修改 1：`rkmpp_encode_close()` 函数（Line 1079-1105）

```c
static av_cold int rkmpp_encode_close(AVCodecContext *avctx)
{
    RKMPPEncContext *r = avctx->priv_data;

    // ✅ [FIX 100.81 DIAG-1] 函数入口
    av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-1] rkmpp_encode_close() ENTER\n");
    av_log(avctx, AV_LOG_ERROR, "   priv_data=%p, cfg_init=%d, async_frames=%d\n",
           r, r->cfg_init, r->async_frames);

    r->cfg_init = 0;
    r->async_frames = 0;

    // ✅ [FIX 100.81 DIAG-2] MPP 配置清理
    av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-2] About to deinit mcfg=%p\n", r->mcfg);
    if (r->mcfg) {
        mpp_enc_cfg_deinit(r->mcfg);
        r->mcfg = NULL;
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-3] mcfg deinit SUCCESS\n");
    } else {
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-3] mcfg is NULL, skipped\n");
    }

    // ✅ [FIX 100.81 DIAG-4] MPP 上下文销毁
    av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-4] About to destroy mpp context (mapi=%p, mctx=%p)\n",
           r->mapi, r->mctx);
    if (r->mapi) {
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-5] Calling mapi->reset()\n");
        r->mapi->reset(r->mctx);
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-6] mapi reset SUCCESS, calling mpp_destroy()\n");

        mpp_destroy(r->mctx);
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-7] mpp_destroy() SUCCESS\n");

        r->mctx = NULL;
    } else {
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-5] mapi is NULL, skipped\n");
    }

    // ✅ [FIX 100.81 DIAG-8] 帧列表清理
    av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-8] About to clear frame list (frame_list=%p)\n",
           r->frame_list);
    clear_frame_list(&r->frame_list);
    av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-9] clear_frame_list() SUCCESS\n");

    // ✅ [FIX 100.81 DIAG-10] 硬件上下文清理
    av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-10] About to unref hw contexts (hwframe=%p, hwdevice=%p)\n",
           r->hwframe, r->hwdevice);
    if (r->hwframe) {
        av_buffer_unref(&r->hwframe);
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-11] hwframe unref SUCCESS\n");
    }
    if (r->hwdevice) {
        av_buffer_unref(&r->hwdevice);
        av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-12] hwdevice unref SUCCESS\n");
    }

    av_log(avctx, AV_LOG_ERROR, "[FIX 100.81 DIAG-13] rkmpp_encode_close() EXIT SUCCESS\n");

    return 0;
}
```

### 修改 2：`clear_frame_list()` 函数（Line 238-265）

```c
static void clear_frame_list(MPPEncFrame **list)
{
    int frame_count = 0;
    av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG] clear_frame_list() START (list=%p, *list=%p)\n",
           list, list ? *list : NULL);

    while (*list) {
        MPPEncFrame *frame = NULL;
        MppFrame mpp_frame = NULL;
        MppBuffer mpp_buf = NULL;

        frame_count++;
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG] Clearing frame #%d (frame=%p)\n",
               frame_count, *list);

        frame = *list;
        *list = (*list)->next;

        // MPP frame 清理
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   mpp_frame=%p\n", frame->mpp_frame);
        mpp_frame = frame->mpp_frame;
        if (mpp_frame) {
            mpp_buf = mpp_frame_get_buffer(mpp_frame);
            av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   mpp_buf=%p\n", mpp_buf);

            if (mpp_buf && mpp_buffer_get_index(mpp_buf) >= 0) {
                av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to mpp_buffer_put()\n");
                mpp_buffer_put(mpp_buf);
                av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   mpp_buffer_put() SUCCESS\n");
            }

            av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to mpp_frame_deinit()\n");
            mpp_frame_deinit(&frame->mpp_frame);
            av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   mpp_frame_deinit() SUCCESS\n");
            frame->mpp_frame = NULL;
        }

        // SEI set 清理
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to free SEI set (datas=%p)\n",
               frame->mpp_sei_set.datas);
        av_freep(&frame->mpp_sei_set.datas);
        frame->mpp_sei_set.count = 0;
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   SEI set freed\n");

        // AVFrame 清理
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to av_frame_free() (frame->frame=%p)\n",
               frame->frame);
        av_frame_free(&frame->frame);
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   av_frame_free() SUCCESS\n");

        // MPPEncFrame 清理
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to free MPPEncFrame (frame=%p)\n", frame);
        av_freep(&frame);
        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   MPPEncFrame freed\n");
    }

    av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG] clear_frame_list() END (cleared %d frames)\n",
           frame_count);
}
```

3. **本地缓存**：Docker容器会保留编译中间产物
