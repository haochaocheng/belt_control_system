## 可能的崩溃点

根据源码分析，崩溃可能发生在：

1. **mpp_destroy(r->mctx)** - MPP 上下文销毁
2. **clear_frame_list()** - 帧列表清理
   - `mpp_frame_deinit()` - MPP 帧清理
   - `av_frame_free()` - AVFrame 释放（可能 double-free）
   - `av_freep(&frame)` - MPPEncFrame 释放（可能 double-free）
3. **av_freep(&avctx->priv_data)** - 私有数据释放

static void clear_frame_list(MPPEncFrame **list)
{
    int frame_count = 0;
    while (*list) {
        MPPEncFrame *frame = NULL;
        MppFrame mpp_frame = NULL;
        MppBuffer mpp_buf = NULL;

        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG] Clearing frame #%d\n", ++frame_count);

        frame = *list;
        *list = (*list)->next;

        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to clear mpp_frame\n");
        mpp_frame = frame->mpp_frame;
        if (mpp_frame) {
            mpp_buf = mpp_frame_get_buffer(mpp_frame);
            if (mpp_buf &&
                mpp_buffer_get_index(mpp_buf) >= 0)
                mpp_buffer_put(mpp_buf);

            av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to mpp_frame_deinit\n");
            mpp_frame_deinit(&frame->mpp_frame);
            av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   mpp_frame_deinit success\n");
            frame->mpp_frame = NULL;

        av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG]   About to free SEI set\n");

    }
    av_log(NULL, AV_LOG_ERROR, "[FIX 100.81 DIAG] Cleared %d frames total\n", frame_count);
}
