#### 3.1.2 键盘导航

```qml
// ✅ 2026-01-24 [设备信息界面重构] Phase 4: 键盘导航
Keys.onLeftPressed: {
    if (currentDeviceIndex > 0) {
        currentDeviceIndex--
        updateDeviceSelection()
    }
}

Keys.onRightPressed: {
    if (currentDeviceIndex < 11) {
        currentDeviceIndex++
        updateDeviceSelection()
    }
}

Keys.onUpPressed: {
    if (currentDeviceIndex >= 4) {
        currentDeviceIndex -= 4
        updateDeviceSelection()
    }
}

Keys.onDownPressed: {
    if (currentDeviceIndex <= 7) {
        currentDeviceIndex += 4
        updateDeviceSelection()
    }
}

Keys.onReturnPressed: {
    openDeviceSettings(currentDeviceIndex)
}

Keys.onEnterPressed: {
    openDeviceSettings(currentDeviceIndex)
}
```

**导航逻辑**：
- **左键**：索引 -1（最小 0）
- **右键**：索引 +1（最大 11）
- **上键**：索引 -4（跳到上一行）
- **下键**：索引 +4（跳到下一行）
- **Enter/Return**：打开参数设置弹窗

#### 3.1.3 辅助函数

```qml
// ✅ 2026-01-24 [设备信息界面重构] Phase 4: 辅助函数
function updateDeviceSelection() {
    console.log("🎯 [Input1Page] 选中设备:", currentDeviceIndex)

    // 更新所有设备的选中状态
    for (var i = 0; i < deviceItems.length; i++) {
        if (deviceItems[i]) {
            deviceItems[i].isSelected = (i === currentDeviceIndex)
        }
    }
}

function openDeviceSettings(index) {
    console.log("⚙️ [Input1Page] 打开设备参数设置:", index)
    settingsDialog.deviceId = index + 1
    settingsDialog.deviceName = deviceItems[index] ? deviceItems[index].deviceName : ""
    settingsDialog.visible = true
}

function selectDevice(index) {
    currentDeviceIndex = index
    updateDeviceSelection()
    input1Page.forceActiveFocus()  // 确保页面获得焦点，以便键盘导航
}
```

#### 3.1.4 设备数组初始化和鼠标交互

```qml
onLoaded: {
    // ... 现有代码 ...

    // ✅ 2026-01-24 [设备信息界面重构] Phase 4: 初始化设备数组和交互
    console.log("🎮 [Input1Page] 初始化设备交互逻辑")

    // 获取所有设备组件
    deviceItems = [
        screenLoader.item.device1,
        screenLoader.item.device2,
        screenLoader.item.device3,
        screenLoader.item.device4,
        screenLoader.item.device5,
        screenLoader.item.device6,
        screenLoader.item.device7,
        screenLoader.item.device8,
        screenLoader.item.device9,
        screenLoader.item.device10,
        screenLoader.item.device11,
        screenLoader.item.device12
    ]

    // 为每个设备添加 MouseArea
    for (var i = 0; i < deviceItems.length; i++) {
        (function(index) {
            var device = deviceItems[index]
            if (device) {
                // 动态创建 MouseArea
                var mouseArea = Qt.createQmlObject(
                    'import QtQuick 2.15; MouseArea { anchors.fill: parent; hoverEnabled: true }',
                    device,
                    "dynamicMouseArea"
                )
                mouseArea.clicked.connect(function() {
                    selectDevice(index)
                })
                mouseArea.onDoubleClicked.connect(function() {
                    openDeviceSettings(index)
                })
            }
        })(i)
    }

    // 初始化第一个设备为选中状态
    updateDeviceSelection()
    console.log("✅ [Input1Page] 设备交互逻辑初始化完成")
}
```

**关键技术**：
- 使用 `Qt.createQmlObject` 动态创建 MouseArea
- 使用闭包 `(function(index) { ... })(i)` 捕获正确的索引
- 连接 `clicked` 和 `doubleClicked` 信号

### 3.2 MyIN_Data.ui.qml 修改

**文件路径**: `src/qml/Input1/Input1Content/MyIN_Data.ui.qml`

#### 3.2.1 选中状态视觉反馈

```qml
// ========== 样式属性 ==========
color: "transparent" // 透明背景
border.color: "transparent"  // 默认无边框
border.width: 0  // 默认无边框
radius: 8  // 圆角

// ✅ 2026-01-24 [设备信息界面重构] Phase 4: 选中状态视觉反馈
// 使用 QML 状态系统（QDS 兼容）
states: [
    State {
        name: "selected"
        when: isSelected
        PropertyChanges {
            target: root
            border.color: "#00FF00"  // 绿色边框
            border.width: 3  // 3px 边框
        }
    },
    State {
        name: "normal"
        when: !isSelected
        PropertyChanges {
            target: root
            border.color: "transparent"
            border.width: 0
        }
    }
]
```

**关键技术**：
- ✅ 使用 QML `State` 系统（QDS 兼容）
- ✅ 避免使用三元运算符（`? :`）
- ✅ 使用 `when` 条件自动切换状态
- ✅ 使用 `PropertyChanges` 修改属性

**为什么不用三元运算符？**
```qml
// ❌ 不兼容 QDS（JavaScript 语法）
border.color: isSelected ? "#00FF00" : "transparent"

// ✅ 兼容 QDS（QML 状态系统）
states: [
    State {
        name: "selected"
        when: isSelected
        PropertyChanges { target: root; border.color: "#00FF00" }
    }
]
```

## 4. 交互流程

### 4.1 键盘导航流程

```
用户按下方向键
    ↓
Keys.onXxxPressed 事件触发
    ↓
更新 currentDeviceIndex
    ↓
调用 updateDeviceSelection()
    ↓
遍历 deviceItems 数组
    ↓
设置每个设备的 isSelected 属性
    ↓
MyIN_Data 的 State 系统自动切换
    ↓
选中设备显示绿色边框
```

### 4.2 鼠标点击流程

```
用户点击设备
    ↓
MouseArea.clicked 信号触发
    ↓
调用 selectDevice(index)
    ↓
更新 currentDeviceIndex
    ↓
调用 updateDeviceSelection()
    ↓
选中设备显示绿色边框
    ↓
页面获得焦点（forceActiveFocus）
```

### 4.3 打开弹窗流程

```
用户双击设备 或 按下 Enter 键
    ↓
MouseArea.doubleClicked 或 Keys.onReturnPressed
    ↓
调用 openDeviceSettings(index)
    ↓
设置弹窗的 deviceId 和 deviceName
    ↓
设置 settingsDialog.visible = true
    ↓
弹窗显示
```

## 5. 技术亮点

### 5.1 QDS 兼容性

✅ **完全兼容 Qt Design Studio**：
- Screen01.ui.qml 和 MyIN_Data.ui.qml 保持纯声明式
- 使用 QML State 系统替代三元运算符

### 5.2 动态 MouseArea 创建

✅ **运行时动态创建交互组件**：
```qml
var mouseArea = Qt.createQmlObject(
    'import QtQuick 2.15; MouseArea { anchors.fill: parent; hoverEnabled: true }',
    device,
    "dynamicMouseArea"
)
mouseArea.clicked.connect(function() {
    selectDevice(index)
})
```

**优势**：
- 不修改 .ui.qml 文件
- 保持 QDS 兼容性
- 灵活添加交互逻辑

### 5.3 闭包捕获索引

✅ **正确捕获循环变量**：
```qml
for (var i = 0; i < deviceItems.length; i++) {
    (function(index) {
        // 使用 index 而不是 i
        mouseArea.clicked.connect(function() {
            selectDevice(index)  // ✅ 正确
        })
    })(i)
}
```

**为什么需要闭包？**
- JavaScript 的循环变量 `i` 是共享的
- 不使用闭包，所有回调都会使用最后的 `i` 值（11）
- 使用闭包，每个回调捕获自己的 `index` 值

### 5.4 焦点管理

✅ **确保键盘导航生效**：
```qml
Item {
    id: input1Page
    focus: true  // 页面默认获得焦点
}

function selectDevice(index) {
    // ...
    input1Page.forceActiveFocus()  // 点击后重新获得焦点
}

onIsActiveChanged: {
    if (isActive) {
        input1Page.forceActiveFocus()  // 页面激活时获得焦点
    }
}
```
