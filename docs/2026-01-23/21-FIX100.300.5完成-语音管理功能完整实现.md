### 2. 文件列表组件

**位置**: Lines 448-577

**关键代码**：

```qml
ListView {
    id: fileListView

    // 绑定到后端文件列表
    model: ListModel {
        id: fileListModel
    }

    // 监听文件列表更新
    Connections {
        target: audioManagementController
        function onFileListUpdated() {
            refreshFileList()
        }
    }

    // 刷新文件列表
    function refreshFileList() {
        fileListModel.clear()
        var files = audioManagementController.getFileList()
        for (var i = 0; i < files.length; i++) {
            fileListModel.append(files[i])
        }
    }

    // 文件项委托
    delegate: Rectangle {
        // 文件名
        Text {
            text: model.fileName || ""
        }

        // 状态标签（6种状态）
        Rectangle {
            color: {
                var status = model.status || 0
                if (status === 0) return "#95a5a6"  // Pending
                if (status === 1) return "#f39c12"  // Recognizing
                if (status === 2) return "#3498db"  // Recognized
                if (status === 3) return "#f39c12"  // Generating
                if (status === 4) return "#2ecc71"  // Generated
                if (status === 5) return "#e74c3c"  // Failed
            }
        }

        // 识别的文本
        Text {
            text: model.recognizedText || "（未识别）"
        }

        // 错误信息
        Text {
            visible: model.errorMessage && model.errorMessage.length > 0
            text: "错误: " + (model.errorMessage || "")
        }
    }
}
```

**状态映射**：
- 0 = Pending（待处理）- 灰色
- 1 = Recognizing（识别中）- 橙色
- 2 = Recognized（已识别）- 蓝色
- 3 = Generating（生成中）- 橙色
- 4 = Generated（已生成）- 绿色
- 5 = Failed（失败）- 红色

### 3. Toast 提示组件

**位置**: Lines 715-782

**关键代码**：

```qml
// Toast 提示组件
Rectangle {
    id: toastMessage
    anchors.horizontalCenter: parent.horizontalCenter
    anchors.bottom: parent.bottom
    anchors.bottomMargin: 100
    opacity: 0
    visible: opacity > 0
    z: 1000

    property string message: ""
    property bool isError: false

    Behavior on opacity {
        NumberAnimation { duration: 300 }
    }

    Timer {
        id: toastTimer
        interval: 3000
        onTriggered: toastMessage.opacity = 0
    }

    function show(msg, error) {
        message = msg
        isError = error || false
        opacity = 1
        toastTimer.restart()
    }
}

// 监听后端错误信号
Connections {
    target: audioManagementController

    function onErrorOccurred(message) {
        toastMessage.show(message, true)
    }

    function onProcessingFinished(success, message) {
        toastMessage.show(message, !success)
    }
}
```

**特性**：
- 自动淡入淡出动画（300ms）
- 3秒后自动消失
- 成功/错误两种样式
- 监听后端信号自动显示

## 🎓 技术要点

### 1. QML 双向绑定模式

**正确做法**：
```qml
// QML → C++（用户修改）
Slider {
    value: TTSConfig.rate(TTSConfigManager.Test)  // 初始化

    onValueChanged: {
        TTSConfig.setRate(TTSConfigManager.Test, value)  // 保存
    }
}
```

**关键点**：
- 使用 `value:` 绑定初始化
- 使用 `onValueChanged` 保存修改
- 使用场景枚举（Test）区分不同场景

### 2. ListView 动态刷新

**正确做法**：
```qml
ListView {
    model: ListModel { id: fileListModel }

    Connections {
        target: audioManagementController
        function onFileListUpdated() {
            refreshFileList()
        }
    }

    function refreshFileList() {
        fileListModel.clear()
        var files = audioManagementController.getFileList()
        for (var i = 0; i < files.length; i++) {
            fileListModel.append(files[i])
        }
    }
}
```

**关键点**：
- 使用 ListModel 作为中间层
- 监听后端信号触发刷新
- 使用 `clear()` + `append()` 更新数据

### 3. Toast 提示组件设计

**关键点**：
- 使用 `opacity` 控制显示/隐藏
- 使用 `Behavior` 添加动画
- 使用 `Timer` 自动消失
- 使用 `z: 1000` 确保在最上层

### 4. 状态颜色映射

**最佳实践**：
```qml
color: {
    var status = model.status || 0
    if (status === 0) return "#95a5a6"  // Pending - 灰色
    if (status === 1) return "#f39c12"  // Recognizing - 橙色
    if (status === 2) return "#3498db"  // Recognized - 蓝色
    if (status === 3) return "#f39c12"  // Generating - 橙色
    if (status === 4) return "#2ecc71"  // Generated - 绿色
    if (status === 5) return "#e74c3c"  // Failed - 红色
    return "#95a5a6"
}
```

**颜色选择原则**：
- 灰色 = 未开始
- 橙色 = 进行中
- 蓝色 = 中间状态
- 绿色 = 成功
- 红色 = 失败
