// 在main.cpp中注册
qmlRegisterSingletonType<TTSConfigManager>(

        Q_UNUSED(engine)

    }

**QML使用**:
```qml
import com.belt.control 1.0

// 设置模型
TTSConfig.setModelIndex(TTSConfigManager.Test, modelComboBox.currentIndex)

TTSConfig.setSpeakerId(TTSConfigManager.Test, speakerIdSpinBox.value)

// 设置语速

// 设置音量
TTSConfig.setVolume(TTSConfigManager.Test, volumeSlider.value)
```

#### 2. AudioManagementController
```qml
// 在main.cpp中注册
AudioManagementController *audioController = new AudioManagementController();
engine.rootContext()->setContextProperty("audioController", audioController);
```

**QML使用**:
```qml
// 扫描目录
audioController.scanDirectory(dirPathInput.text)

// 批量识别
audioController.recognizeAll()

// 批量生成
audioController.generateAllTts()

// 停止处理
audioController.stopProcessing()

// 绑定属性
ProgressBar {
    value: audioController.progress / 100
}

Text {
    text: audioController.processedFiles + " / " + audioController.totalFiles
}
```
