## 🔍 IP 地址提取逻辑

### PowerShell 实现

```powershell
# 1. 获取所有 IPv4 地址（排除回环和自动配置地址）
$localIPs = Get-NetIPAddress -AddressFamily IPv4 | Where-Object {
    $_.IPAddress -notlike "127.*" -and      # 排除回环
    $_.IPAddress -notlike "169.254.*"       # 排除自动配置
}

# 2. 优先选择 192.168.x.x 网段
$targetIP = $localIPs | Where-Object { $_.IPAddress -like "192.168.*" } | Select-Object -First 1

# 3. 如果没有，选择第一个非回环地址
if (-not $targetIP) {
    $targetIP = $localIPs | Select-Object -First 1
}

# 4. 解析第3字节
$ipBytes = $targetIP.IPAddress.Split('.')
$thirdByte = [int]$ipBytes[2]

# 5. 计算组播地址
$multicastAddr = "224.1.$thirdByte.1"
```

### C++/Qt 实现（待实现）

```cpp
QHostAddress AudioNetworkSender::getLocalIPAddress()
{
    // 获取所有网络接口
    const QList<QHostAddress> addresses = QNetworkInterface::allAddresses();

    // 优先查找 192.168.x.x 网段的地址
    for (const QHostAddress& address : addresses) {
        if (address.protocol() == QAbstractSocket::IPv4Protocol &&
            !address.isLoopback() &&
            address.toString().startsWith("192.168.")) {
            return address;
        }
    }

    // 如果没找到，返回第一个非回环地址
    for (const QHostAddress& address : addresses) {
        if (address.protocol() == QAbstractSocket::IPv4Protocol &&
            !address.isLoopback()) {
            return address;
        }
    }

    // 兜底：返回 localhost
    qWarning() << "⚠️ 无法获取本地 IP，使用 127.0.0.1";
    return QHostAddress::LocalHost;
}
```
