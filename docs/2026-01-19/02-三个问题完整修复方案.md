## 🔧 修复 1：Opus 16kHz 配置

### 根本原因
- PJSIP Opus 默认配置是 48kHz/2
- `codecSetPriority("opus/16000/1", 213)` 只设置了不存在的编解码器优先级

            pjmedia_codec_param opus_param;

            // 获取当前配置
            status = pjmedia_codec_opus_get_config(&opus_cfg);
            if (status != PJ_SUCCESS) {
                qWarning() << "  ⚠️ Failed to get Opus config, using default";
                // 初始化默认值
                opus_cfg.sample_rate = 16000;
                opus_cfg.channel_cnt = 1;

                qDebug() << "  📊 Original Opus config: " << opus_cfg.sample_rate << "Hz, " << opus_cfg.channel_cnt << " ch";

                opus_cfg.channel_cnt = 1;        // Mono (VoIP standard)

                pjmedia_endpt_get_codec_mgr(med_endpt),
                nullptr,
                &opus_param
            );

                char errmsg[PJ_ERR_MSG_SIZE];
                pj_strerror(status, errmsg, sizeof(errmsg));
                qWarning() << "  ⚠️ Failed to set Opus default param:" << errmsg;
            } else {
                qInfo() << "  ✅ Opus configured: 16kHz mono (wideband audio)";
            }
        }
    } else {
        qWarning() << "  ⚠️ Media endpoint not ready, skipping Opus initialization";

void SipPhoneManager::setMicrophoneVolume(int volume)
{
    qDebug() << "🎤 [Volume] Setting microphone volume:" << volume << "%";

    // ✅ 保存到 QSettings（下次启动恢复）
    QSettings settings;
    settings.setValue("SIP/MicrophoneVolume", volume);
    qDebug() << "  ✅ [Settings] Microphone volume saved:" << volume;

    // ✅ 如果 PJSIP 已初始化，立即应用音量设置
    if (d->initialized) {
        // 转换百分比 (0-100) 到 PJSIP 音量 (0-255)
        unsigned int pj_volume = (volume * 255) / 100;

        // 使用 PJSIP API 设置麦克风增益
        pj_status_t status = pjsua_snd_set_setting(
            PJMEDIA_AUD_DEV_CAP_INPUT_VOLUME_SETTING,
            &pj_volume,
            PJ_TRUE  // Keep setting for future devices
        );

        if (status == PJ_SUCCESS) {
            qDebug() << "  ✅ [PJSIP] Microphone volume applied (PJSIP level:" << pj_volume << "/255)";
        } else {
            char errmsg[PJ_ERR_MSG_SIZE];
            pj_strerror(status, errmsg, sizeof(errmsg));
            qWarning() << "  ⚠️ [PJSIP] Failed to set mic volume:" << errmsg;
            qWarning() << "     But saved to settings, will be used when audio device is opened";

        qDebug() << "  📌 [PJSIP] Not initialized yet, volume will be applied when PJSIP starts";

## 🔧 修复 3：麦克风选择配置保存问题

### 问题分析
1. ✅ QSettings 保存/读取代码**已存在**
2. ⚠️ **设备索引映射错误**（CLAUDE.md 记录的问题）：
   - 用户层设备索引（过滤后）：0, 1, 2...
   - PJSIP 层设备索引（原始）：0, 1, 2, ..., 28
   - 如果保存的是用户层索引，但 PJSIP 使用原始索引 → 设备错误！

### 解决方案：设备索引映射数组

**文件**: `src/sip_phone/SipPhoneManager.h`

QStringList SipPhoneManager::enumerateAudioInputDevices()
{
    QStringList devices;

    // ✅ 2026-01-19 05:00 清空映射数组
    m_audioInputDeviceMapping.clear();

    unsigned aud_dev_count = pjmedia_aud_dev_count();
    qDebug() << "📢 [Enumerate] Total audio devices:" << aud_dev_count;

    for (unsigned i = 0; i < aud_dev_count; ++i) {
        pjmedia_aud_dev_info dev_info;
        pj_status_t status = pjmedia_aud_dev_get_info(i, &dev_info);
        if (status == PJ_SUCCESS) {
            QString dev_name = QString::fromUtf8(dev_info.name);

            // 过滤逻辑（与现有代码一致）
            if (dev_info.input_count > 0 &&
                !dev_name.contains("Loopback", Qt::CaseInsensitive) &&
                !dev_name.contains("default", Qt::CaseInsensitive)) {

                devices << dev_name;

                // ✅ 2026-01-19 05:00 保存 PJSIP 原始索引
                m_audioInputDeviceMapping.append(i);

                qDebug() << "  [Mapping] User index:" << (devices.size() - 1)
                         << "→ PJSIP index:" << i
                         << "Device:" << dev_name;
            }
        }
    }

    qDebug() << "📢 [Enumerate] Filtered devices:" << devices.size();
    return devices;
}

bool SipPhoneManager::setAudioInputDevice(int userIndex)
{
    qDebug() << "📢 [FIX] Setting audio input device to user index:" << userIndex;

    // ✅ 2026-01-19 05:00 索引映射：用户索引 → PJSIP 索引
    if (userIndex < 0 || userIndex >= m_audioInputDeviceMapping.size()) {
        qWarning() << "❌ Invalid user device index:" << userIndex;
        return false;
    }

    int pjsipIndex = m_audioInputDeviceMapping[userIndex];
    qDebug() << "  [Mapping] User index:" << userIndex << "→ PJSIP index:" << pjsipIndex;

    // ✅ 禁止在通话中切换设备
    if (d->inCall) {
        qWarning() << "⚠️ Cannot change audio device during active call!";
        return false;
    }

    // ✅ 保存用户索引到 QSettings（不是 PJSIP 索引！）
    QSettings settings;
    settings.setValue("SIP/AudioInputDeviceUserIndex", userIndex);
    qDebug() << "✅ [Settings] Saved user device index:" << userIndex;

    // ✅ 如果 PJSIP 已初始化，使用 PJSIP 索引应用设置
    if (d->initialized) {
        qDebug() << "   [PJSIP] Applying device using PJSIP index:" << pjsipIndex;

        pjsua_snd_dev_param param;
        pjsua_snd_dev_param_default(&param);
        pj_status_t status = pjsua_get_snd_dev2(&param);

        if (status != PJ_SUCCESS) {
            param.playback_dev = PJSUA_SND_DEFAULT_PLAYBACK_DEV;
        }

        // ⚠️ 使用 PJSIP 索引
        param.capture_dev = pjsipIndex;

        status = pjsua_set_snd_dev2(&param);

        if (status == PJ_SUCCESS) {
            qDebug() << "✅ Audio input device changed (PJSIP index:" << pjsipIndex << ")";
            return true;
        } else {
            char errmsg[PJ_ERR_MSG_SIZE];
            pj_strerror(status, errmsg, sizeof(errmsg));
            qWarning() << "❌ Failed to set audio input device:" << errmsg;
            return false;
        }
    } else {
        qDebug() << "   [PJSIP] Not initialized, device will be used when PJSIP starts";
        return true;
    }
}
