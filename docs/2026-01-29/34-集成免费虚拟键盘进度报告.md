   ```qml
   function sendKeyToTarget(keyText) {
       if (keyText === "\x7F") {
           // 退格键处理
           if (targetField.hasOwnProperty("text")) {
               targetField.text = targetField.text.slice(0, -1)
           } else if (targetField.hasOwnProperty("value")) {
               // SpinBox 处理
               var str = targetField.value.toString()
               targetField.value = parseInt(str.slice(0, -1) || "0")
           }
       } else {
           // 普通字符处理
           if (targetField.hasOwnProperty("text")) {
               targetField.text += keyText
           } else if (targetField.hasOwnProperty("value")) {
               // SpinBox 处理
               targetField.value = parseInt(targetField.value.toString() + keyText)
           }
       }
   }
   ```

4. **替换 InputEngine 调用**：
   ```qml
   // 原始：InputEngine.sendKeyToFocusItem(text)
   // 修改：sendKeyToTarget(text)
   ```
