```qml
// ✅ 2026-02-04 [FIX 100.300.113 Phase 7.37 - Phase 4]: 修改 getParamFieldCount 函数
// 返回当前 Tab 的参数数量（动态获取）
function getParamFieldCount() {
    if (!serialConfigPanel.item) {
        console.warn("⚠️ [SerialPortControlPage] SerialPortConfigPanel 未加载")
        return 0
    }

    var currentTab = serialConfigPanel.item.getCurrentTab()
    if (!currentTab) {
        console.warn("⚠️ [SerialPortControlPage] 当前 Tab 未加载")
        return 0
    }

    if (typeof currentTab.getParamFieldCount === "function") {
        return currentTab.getParamFieldCount()
    }

    // 默认参数数量（根据 Tab 索引）
    switch(root.focusTabIndex) {
    case 0:  // 参数配置
        return 10
    case 1:  // 发送区
        return 5
    case 2:  // 接收区
        return 5
    case 3:  // MODBUS寄存器
        return 8
    default:
        return 0
    }
}
```

**效果**:
- ✅ 动态获取当前 Tab 的参数数量
- ✅ 优先调用 Tab 的 getParamFieldCount() 方法
- ✅ 如果 Tab 没有该方法，使用默认值

## 💡 技术要点

### NavigationManager 的 skipTabArea

**为什么设置 skipTabArea = false**:
```qml
skipTabArea = false  // ✅ 支持 Tab 导航
```

**效果**:
- ✅ 列表 → Tab → 参数 → 按钮（四层导航）
- ✅ 如果 skipTabArea = true，则跳过 Tab 区域（列表 → 参数 → 按钮）

### 动态参数数量

**为什么使用 Qt.callLater()**:
```qml
Qt.callLater(function() {
    var paramCount = root.getParamFieldCount()
    updateLastParamIndex(paramCount)
})
```

**原因**:
- ✅ Tab 切换时，Loader 可能还没有加载完成
- ✅ Qt.callLater() 确保在下一个事件循环中执行
- ✅ 此时 Loader 已经加载完成，可以正确获取参数数量

### 信号转发

**为什么使用信号转发**:
```qml
item.requestFocusParamIndex.connect(function(paramIndex) {
    root.focusParamIndex = paramIndex
    navigationManager.paramIndex = paramIndex
})
```

**原因**:
- ✅ 避免直接修改 root.focusParamIndex（破坏 Qt.binding）
- ✅ 通过信号转发，保持单向数据流
- ✅ NavigationManager 是唯一的焦点状态源

### Qt.binding() 的使用

**为什么使用 Qt.binding()**:
```qml
// ❌ 错误：直接赋值会破坏双向绑定
item.focusTabIndex = root.focusTabIndex

// ✅ 正确：使用 Qt.binding() 保持双向绑定
item.focusTabIndex = Qt.binding(function() {
    return root.focusTabIndex
})
```

**效果**:
- ✅ 当 root.focusTabIndex 变化时，item.focusTabIndex 自动更新
- ✅ 保持双向绑定，不会破坏 QML 的响应式系统
