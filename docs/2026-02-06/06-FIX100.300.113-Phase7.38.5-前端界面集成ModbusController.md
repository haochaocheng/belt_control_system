**集成内容**:

- ✅ 调用 modbusController.writeSingleRegister() 写单个寄存器
- ✅ 调用 modbusController.writeMultipleRegisters() 写多个寄存器
- ✅ 监听读写完成信号，动态更新寄存器列表
- ✅ 支持 HEX/DEC 双格式显示

**关键代码**:

```qml
// 读取按钮
Button {
    id: readButton
    text: "读取"
    enabled: functionCode.currentIndex === 0

    onClicked: {
        console.log("✅ [ModbusRegisterTab] 读取寄存器")
        console.log("   - 从站地址:", slaveAddress.text)
        console.log("   - 起始地址:", startAddress.text)
        console.log("   - 数量:", quantity.value)

        // 调用 modbusController 读取寄存器
        modbusController.readHoldingRegisters(
            parseInt(slaveAddress.text, 16),
            parseInt(startAddress.text, 16),
            quantity.value
        )
    }
}

// 写入按钮
Button {
    id: writeButton
    text: "写入"
    enabled: functionCode.currentIndex > 0 && writeValue.text.length > 0

    onClicked: {
        console.log("✅ [ModbusRegisterTab] 写入寄存器")
        console.log("   - 从站地址:", slaveAddress.text)
        console.log("   - 起始地址:", startAddress.text)
        console.log("   - 写入值:", writeValue.text)

        if (functionCode.currentIndex === 1) {
            // 06-写单个寄存器
            modbusController.writeSingleRegister(
                parseInt(slaveAddress.text, 16),
                parseInt(startAddress.text, 16),
                parseInt(writeValue.text, 16)
            )
        } else {
            // 10-写多个寄存器
            modbusController.writeMultipleRegisters(
                parseInt(slaveAddress.text, 16),
                parseInt(startAddress.text, 16),
                [parseInt(writeValue.text, 16)]
            )
        }
    }
}

// 监听 modbusController 信号
Connections {
    target: modbusController

    // 监听读取寄存器完成
    function onReadRegistersFinished(serverAddress, startAddress, values) {
        console.log("✅ [ModbusRegisterTab] 读取寄存器完成")
        console.log("   - 从站地址:", serverAddress)
        console.log("   - 起始地址:", startAddress)
        console.log("   - 数量:", values.length)

        // 清空现有数据
        registerModel.clear()

        // 添加新数据
        for (var i = 0; i < values.length; i++) {
            var addr = startAddress + i
            var hexValue = values[i].toString(16).toUpperCase().padStart(4, '0')
            var decValue = values[i].toString()
            registerModel.append({
                address: addr.toString(16).toUpperCase().padStart(4, '0'),
                hexValue: hexValue,
                decValue: decValue,
                description: "寄存器 " + addr
            })
        }
    }

    // 监听写入寄存器完成
    function onWriteRegisterFinished(serverAddress, address, success) {
        if (success) {
            console.log("✅ [ModbusRegisterTab] 写入单个寄存器成功")
        } else {
            console.error("❌ [ModbusRegisterTab] 写入单个寄存器失败")
        }
    }

    // 监听写入多个寄存器完成
    function onWriteRegistersFinished(serverAddress, startAddress, success) {
        if (success) {
            console.log("✅ [ModbusRegisterTab] 写入多个寄存器成功")
        } else {
            console.error("❌ [ModbusRegisterTab] 写入多个寄存器失败")
        }
    }

    // 监听错误信号
    function onErrorOccurred(error) {
        console.error("❌ [ModbusRegisterTab] MODBUS 错误:", error)
        // TODO: 显示错误提示
    }
}
```

## 🔧 技术细节

### 数据流向

#### 读取寄存器流向
```
用户点击读取按钮
    ↓
modbusController.readHoldingRegisters(serverAddress, startAddress, count)
    ↓
ModbusController::readHoldingRegisters()
    ↓
QModbusRtuSerialMaster::sendReadRequest()
    ↓
MODBUS RTU 请求（功能码 0x03）
    ↓
从站响应
    ↓
QModbusReply::finished 信号
    ↓
readRegistersFinished 信号
    ↓
onReadRegistersFinished()
    ↓
registerModel.clear() + append()
    ↓
ListView 更新显示
```

#### 写入寄存器流向
```
用户输入值并点击写入按钮
    ↓
modbusController.writeSingleRegister() 或 writeMultipleRegisters()
    ↓
ModbusController::writeSingleRegister() 或 writeMultipleRegisters()
    ↓
QModbusRtuSerialMaster::sendWriteRequest()
    ↓
MODBUS RTU 请求（功能码 0x06 或 0x10）
    ↓
从站响应
    ↓
QModbusReply::finished 信号
    ↓
writeRegisterFinished 或 writeRegistersFinished 信号
    ↓
onWriteRegisterFinished() 或 onWriteRegistersFinished()
    ↓
日志输出（成功/失败）
```

### HEX 地址转换

**问题**: 用户输入的是 HEX 字符串，需要转换为整数

**解决方案**:
```qml
// 从站地址：01 → 1
parseInt(slaveAddress.text, 16)

// 起始地址：0000 → 0
parseInt(startAddress.text, 16)

// 写入值：00C8 → 200
parseInt(writeValue.text, 16)
```

### 寄存器列表动态更新

**实现方式**:
```qml
// 清空现有数据
registerModel.clear()

// 添加新数据
for (var i = 0; i < values.length; i++) {
    var addr = startAddress + i
    var hexValue = values[i].toString(16).toUpperCase().padStart(4, '0')
    var decValue = values[i].toString()
    registerModel.append({
        address: addr.toString(16).toUpperCase().padStart(4, '0'),
        hexValue: hexValue,
        decValue: decValue,
        description: "寄存器 " + addr
    })
}
```

**显示格式**:
- 地址：HEX 格式，4位，前导0（例如：0000, 0001, 00C8）
- 值(HEX)：HEX 格式，4位，前导0（例如：0001, 0064, 00C8）
- 值(DEC)：十进制格式（例如：1, 100, 200）
- 说明：自动生成（例如：寄存器 0, 寄存器 1）

### 功能码切换

**实现方式**:
```qml
DeviceInfo.CustomComboBox {
```

## 📝 注意事项

1. **地址格式**：
   - 用户输入：HEX 字符串（例如：01, 0000, 00C8）
   - 后端接收：整数（例如：1, 0, 200）
   - 使用 parseInt(text, 16) 转换

2. **功能码切换**：
   - 03-读保持寄存器：只启用读取按钮
   - 06-写单个寄存器：启用写入按钮和写入值输入框
   - 10-写多个寄存器：启用写入按钮和写入值输入框

3. **寄存器列表**：
   - 读取完成后自动清空并更新
   - 支持 HEX/DEC 双格式显示
   - 地址和值都是 4 位 HEX，前导 0

4. **错误处理**：
   - 所有操作都监听 errorOccurred 信号
   - 目前只打印日志，后续需要添加 UI 提示

5. **MODBUS 连接**：
   - 需要先连接 MODBUS 设备才能读写寄存器
   - 连接状态由 modbusController.isConnected 控制
   - 底部按钮区域需要添加连接/断开按钮
