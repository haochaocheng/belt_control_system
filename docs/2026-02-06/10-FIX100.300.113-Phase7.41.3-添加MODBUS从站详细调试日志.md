// ✅ 2026-02-06 [FIX 100.300.113 Phase 7.41.3]: 串口数据接收调试槽函数
void ModbusSlaveController::handleSerialPortReadyRead()
{
    QSerialPort *serialPort = qobject_cast<QSerialPort*>(sender());
    if (!serialPort) {
        return;
    }

    QByteArray data = serialPort->readAll();
    qDebug() << "🔍 [ModbusSlaveController] 串口接收到数据 - 字节数:" << data.size();
    qDebug() << "   - 原始数据(HEX):" << data.toHex(' ');

    // 解析 MODBUS RTU 帧（简单解析，用于调试）
    if (data.size() >= 4) {
        quint8 slaveAddr = static_cast<quint8>(data[0]);
        quint8 functionCode = static_cast<quint8>(data[1]);

        qDebug() << "   - 从站地址:" << slaveAddr << "(期望:" << m_slaveAddress << ")";
        qDebug() << "   - 功能码:" << QString("0x%1").arg(functionCode, 2, 16, QChar('0'));

        if (slaveAddr != m_slaveAddress) {
            qWarning() << "⚠️ [ModbusSlaveController] 从站地址不匹配！";
        }

        // 解析功能码
        switch (functionCode) {
        case 0x01:
            qDebug() << "   - 功能码: 读取线圈 (0x01)";
            break;
        case 0x03:
            qDebug() << "   - 功能码: 读取保持寄存器 (0x03)";
            break;
        case 0x06:
            qDebug() << "   - 功能码: 写入单个保持寄存器 (0x06)";
            break;
        // ... 其他功能码
        }

        // 如果数据足够长，解析起始地址和数量
        if (data.size() >= 6) {
            quint16 startAddr = (static_cast<quint8>(data[2]) << 8) | static_cast<quint8>(data[3]);
            quint16 quantity = (static_cast<quint8>(data[4]) << 8) | static_cast<quint8>(data[5]);
            qDebug() << "   - 起始地址:" << startAddr;
            qDebug() << "   - 数量:" << quantity;
        }
    } else {
        qWarning() << "⚠️ [ModbusSlaveController] 接收到的数据太短，无法解析 MODBUS 帧";
    }
}
```

void ModbusSlaveController::handleSerialPortError(QSerialPort::SerialPortError error)
{
    if (error == QSerialPort::NoError) {
        return;
    }

    QSerialPort *serialPort = qobject_cast<QSerialPort*>(sender());
    QString errorString = serialPort ? serialPort->errorString() : "未知错误";

    qWarning() << "❌ [ModbusSlaveController] 串口错误:" << error << "-" << errorString;

    // 详细错误类型
    switch (error) {
    case QSerialPort::DeviceNotFoundError:
        qWarning() << "   - 错误类型: 设备未找到";
        break;
    case QSerialPort::PermissionError:
        qWarning() << "   - 错误类型: 权限错误";
        break;
    case QSerialPort::ReadError:
        qWarning() << "   - 错误类型: 读取错误";
        break;
    case QSerialPort::ResourceError:
        qWarning() << "   - 错误类型: 资源错误（设备可能被拔出）";
        break;
    // ... 其他错误类型
    }
}

void ModbusSlaveController::handleDataWritten(QModbusDataUnit::RegisterType table, int address, int size)
{
    qDebug() << "✅ [ModbusSlaveController] 数据被写入 - 表类型:" << table << "地址:" << address << "数量:" << size;

    // 根据表类型发送相应的信号
    switch (table) {
    case QModbusDataUnit::HoldingRegisters:
        qDebug() << "   - 保持寄存器被写入";
        for (int i = 0; i < size; ++i) {
            int value = getHoldingRegister(address + i);
            qDebug() << "     地址" << (address + i) << "值:" << value;
            emit holdingRegisterWritten(address + i, value);
        }
        break;
    case QModbusDataUnit::Coils:
        qDebug() << "   - 线圈被写入";
        for (int i = 0; i < size; ++i) {
            bool value = getCoil(address + i);
            qDebug() << "     地址" << (address + i) << "值:" << value;
            emit coilWritten(address + i, value);
        }
        break;
    default:
        qDebug() << "   - 其他类型被写入";
        break;
    }
}
```

## 🎯 调试日志的作用

### 1. 确认串口是否接收到数据
- 如果看到 `🔍 [ModbusSlaveController] 串口接收到数据` 日志
  - 说明串口确实接收到了数据
  - 问题可能在 MODBUS 协议解析层面
- 如果没有看到这个日志
  - 说明串口根本没有接收到数据
  - **问题在物理连接或串口选择层面**

### 2. 检查从站地址是否匹配
- 日志会显示：`从站地址: X (期望: Y)`
- 如果不匹配，会有警告：`⚠️ [ModbusSlaveController] 从站地址不匹配！`

### 3. 识别 MODBUS 功能码
- 日志会显示功能码的含义
- 例如：`功能码: 读取保持寄存器 (0x03)`

### 4. 查看请求的地址和数量
- 日志会显示：`起始地址: 0` 和 `数量: 10`
- 帮助理解主站的请求内容

### 5. 监控串口错误
- 如果串口出现错误，会立即打印详细错误信息
- 例如：`❌ [ModbusSlaveController] 串口错误: 权限错误`

## ⚠️ 重要发现：串口接收区可能影响 RTU 协议

### 用户反馈
> "在串口参数区有一个接受参数区，会显示接受到的数据，是否影响rtu协议"

### 问题分析

**可能的冲突场景**：
1. **SerialPortController 和 ModbusSlaveController 同时监听串口**
   - SerialPortController 的接收区会读取串口数据
   - ModbusSlaveController 也需要读取串口数据
   - 如果两者同时读取，可能导致数据被分割或丢失

2. **数据竞争**：
   - 串口数据到达时，可能先被 SerialPortController 读取
   - ModbusSlaveController 就无法接收到完整的 MODBUS 帧
   - 导致 MODBUS 协议解析失败

3. **串口占用**：
   - 虽然 Phase 7.41.1 已经修复了串口打开的冲突
   - 但可能存在数据读取的竞争

### 解决方案

**方案 1：启动从站前禁用接收区**
- 在启动从站时，停止 SerialPortController 的数据接收
- 确保只有 ModbusSlaveController 读取串口数据

**方案 2：使用不同的串口**
- 主站和从站使用不同的串口
- 避免数据竞争

**方案 3：添加数据读取互斥锁**
- 使用互斥锁确保同一时间只有一个对象读取串口数据
- 但这会增加复杂度

### 建议测试步骤

1. **测试 1：关闭接收区后启动从站**
   - 在"接收区" Tab 中，停止数据接收
   - 然后启动从站
   - 查看是否能接收到数据

2. **测试 2：使用不同的串口**
   - 主站连接到 COM5
   - 从站也启动在 COM5
   - 查看日志是否显示接收到数据

3. **测试 3：查看调试日志**
   - 启动从站后，使用主站工具发送请求
   - 查看是否有 `🔍 [ModbusSlaveController] 串口接收到数据` 日志
   - 如果没有，说明数据被其他地方读取了

## 💡 经验总结

### 成功经验
1. ✅ 使用 `findChild<QSerialPort*>()` 获取底层串口对象
2. ✅ 使用 `Qt::UniqueConnection` 避免重复连接信号
3. ✅ 详细的日志输出，包括原始字节流和解析结果
4. ✅ 识别用户反馈的潜在问题（串口接收区冲突）

### 技术要点
1. **QModbusRtuSerialServer 内部使用 QSerialPort**
   - 可以通过 `findChild<QSerialPort*>()` 获取
   - 可以连接串口的信号进行调试

2. **MODBUS RTU 帧格式**
   - 字节 0: 从站地址
   - 字节 1: 功能码
   - 字节 2-3: 起始地址（大端序）
   - 字节 4-5: 数量（大端序）
   - 最后 2 字节: CRC 校验

3. **串口数据竞争**
   - 多个对象同时读取串口会导致数据分割
   - 需要确保同一时间只有一个对象读取串口
