```cpp
#ifndef TTSBATCHCONFIG_H
#define TTSBATCHCONFIG_H
```

```cpp
// ✅ 2026-02-24 [Phase 7.47.2]: TTS 批量生成器

#ifndef TTSBATCHGENERATOR_H
#define TTSBATCHGENERATOR_H

#include <QObject>
#include <QThread>
#include "TTSBatchConfig.h"
#include "TTSEngineManager.h"

// 生成任务项
struct TTSGenerationTask {
    QString text;                // 要合成的文本
    QString outputPath;          // 输出文件路径
    TTSEngineConfig engineConfig; // 引擎配置
};

// 生成进度信息
struct TTSGenerationProgress {
    int totalFiles;              // 总文件数
    int completedFiles;          // 已完成文件数
    int successFiles;            // 成功文件数
    int failedFiles;             // 失败文件数
    int skippedFiles;            // 跳过文件数
    QString currentFile;         // 当前正在生成的文件
    QString currentEngine;       // 当前使用的引擎
    qint64 elapsedTime;          // 已用时间（毫秒）
    qint64 estimatedTimeLeft;    // 预计剩余时间（毫秒）
};

class TTSBatchGenerator : public QObject
{
    Q_OBJECT

public:
    explicit TTSBatchGenerator(QObject *parent = nullptr);
    ~TTSBatchGenerator();

    // 设置配置
    void setConfig(const TTSBatchConfig &config);

    // 开始生成
    void start();

    // 停止生成
    void stop();

    // 暂停/恢复
    void pause();
    void resume();

    // 获取当前进度
    TTSGenerationProgress getProgress() const;

signals:
    // 进度更新信号
    void progressUpdated(const TTSGenerationProgress &progress);

    // 单个文件完成信号
    void fileCompleted(const QString &filePath, bool success, const QString &error);

    // 全部完成信号
    void allCompleted(int totalFiles, int successFiles, int failedFiles);

    // 错误信号
    void errorOccurred(const QString &error);

private:
    // 生成任务列表
    QVector<TTSGenerationTask> generateTaskList();

    // 生成单个文件
    bool generateSingleFile(const TTSGenerationTask &task, QString &error);

    // 工作线程
    QThread *m_workerThread;
    TTSBatchConfig m_config;
    TTSEngineManager *m_engineManager;

    // 状态
    bool m_running;
    bool m_paused;
    bool m_stopRequested;

    // 进度
    TTSGenerationProgress m_progress;
};

#endif // TTSBATCHGENERATOR_H
```

### 2.3 实现任务列表生成逻辑

**文件**: `src/control/tts/TTSBatchGenerator.cpp`

关键方法：
```cpp
QVector<TTSGenerationTask> TTSBatchGenerator::generateTaskList()
{
    QVector<TTSGenerationTask> tasks;

    // 遍历每个引擎配置
    for (const auto &engineConfig : m_config.engines) {
        QString engineFolder = m_config.outputBaseDir + "/" + engineConfig.outputFolder;

        // 遍历每个类别
        for (const auto &category : m_config.categories) {
            switch (category) {
                case ProtectionCategory::SwitchInput:
                    tasks.append(generateSwitchInputTasks(engineConfig, engineFolder));
                    break;
                case ProtectionCategory::AnalogInput:
                    tasks.append(generateAnalogInputTasks(engineConfig, engineFolder));
                    break;
                case ProtectionCategory::Motor:
                    tasks.append(generateMotorTasks(engineConfig, engineFolder));
                    break;
                case ProtectionCategory::Brake:
                    tasks.append(generateBrakeTasks(engineConfig, engineFolder));
                    break;
                case ProtectionCategory::Tension:
                    tasks.append(generateTensionTasks(engineConfig, engineFolder));
                    break;
                case ProtectionCategory::LinePosition:
                    tasks.append(generateLinePositionTasks(engineConfig, engineFolder));
                    break;
                case ProtectionCategory::SystemSound:
                    tasks.append(generateSystemSoundTasks(engineConfig, engineFolder));
                    break;
            }
        }
    }

    return tasks;
}

// 生成开关量输入保护任务
QVector<TTSGenerationTask> TTSBatchGenerator::generateSwitchInputTasks(
    const TTSEngineConfig &engineConfig, const QString &baseFolder)
{
    QVector<TTSGenerationTask> tasks;
    QStringList protectionTypes = {"沿线急停", "沿线跑偏", "沿线撕裂", "烟雾", "温度", "护网", "堆煤", "主机急停"};

    for (int beltNum : m_config.beltNumbers) {
        QString beltFolder = baseFolder + QString("/%1#PD").arg(beltNum);

        for (const QString &protection : protectionTypes) {
            TTSGenerationTask task;
            task.text = QString("%1号皮带%2保护").arg(beltNum).arg(protection);
            task.outputPath = beltFolder + "/" + task.text + ".wav";
            task.engineConfig = engineConfig;
            tasks.append(task);
        }
    }

    return tasks;
}
```

```qml
Dialog {
    id: root
```

```qml
    contentItem: ColumnLayout {

            title: "📝 当前任务"

                wrapMode: Text.Wrap
            }
        }

        // 统计信息

            title: "📈 统计信息"
            Layout.fillWidth: true

            GridLayout {
                columns: 2
                Label { text: "✅ 成功:" }
                Label { text: progress ? progress.successFiles : 0 }
                Label { text: "❌ 失败:" }
                Label { text: progress ? progress.failedFiles : 0 }
                Label { text: "⏭️ 跳过:" }
                Label { text: progress ? progress.skippedFiles : 0 }
            }
        }

        // 日志输出
        GroupBox {
            title: "📋 日志"
            Layout.fillWidth: true
            Layout.fillHeight: true

            ScrollView {
                anchors.fill: parent
                TextArea {
                    id: logArea
                    readOnly: true
                    wrapMode: TextArea.Wrap
                }
            }
        }
    }

    footer: DialogButtonBox {
        Button {
            text: "⏹️ 停止"
            onClicked: {
                if (ttsGenerator) {
                    ttsGenerator.stop()
                }
            }
        }
        Button {
            text: "关闭"
            enabled: progress && progress.completedFiles === progress.totalFiles
            DialogButtonBox.buttonRole: DialogButtonBox.AcceptRole
        }
    }

    function formatTime(ms) {
        var seconds = Math.floor(ms / 1000)
        var minutes = Math.floor(seconds / 60)
        var hours = Math.floor(minutes / 60)

        if (hours > 0) {
            return hours + " 小时 " + (minutes % 60) + " 分钟"
        } else if (minutes > 0) {
            return minutes + " 分钟 " + (seconds % 60) + " 秒"
        } else {
            return seconds + " 秒"
        }
    }

    Connections {
        target: ttsGenerator

        function onProgressUpdated(progress) {
            root.progress = progress
        }

        function onFileCompleted(filePath, success, error) {
            var status = success ? "✅" : "❌"
            logArea.append(status + " " + filePath + (error ? " - " + error : ""))
        }
    }
}
```
