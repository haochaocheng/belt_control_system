#### 1.3 添加 Keys.onPressed 处理器（Line 287-338）

```qml
// ✅ 2026-01-31 [FIX 100.300.112.8]: 键盘导航支持
focus: true

Keys.onPressed: (event) => {
    var direction = ""

    switch(event.key) {
    case Qt.Key_Up:
        direction = "Up"
        break
    case Qt.Key_Down:
        direction = "Down"
        break
    case Qt.Key_Left:
        direction = "Left"
        break
    case Qt.Key_Right:
        direction = "Right"
        break
    case Qt.Key_Return:
    case Qt.Key_Enter:
        // 回车键：触发当前焦点项
        if (navigationManager.currentArea === navigationManager.areaUsageStatus) {
            tensionControlConfigPanel.item.toggleUsageStatus()
        } else if (navigationManager.currentArea === navigationManager.areaParams) {
            tensionControlConfigPanel.item.triggerParamInput(navigationManager.paramIndex)
        } else if (navigationManager.currentArea === navigationManager.areaButtons) {
            tensionControlConfigPanel.item.triggerButton(navigationManager.buttonIndex)
        }
        event.accepted = true
        return
    default:
        return
    }

    // 根据当前区域调用对应的导航函数
    switch(navigationManager.currentArea) {
    case navigationManager.areaControlList:
        navigationManager.moveInListArea(direction)
        break
    case navigationManager.areaUsageStatus:
        navigationManager.moveInUsageStatusArea(direction)
        break
    case navigationManager.areaParams:
        navigationManager.moveInParamArea(direction)
        break
    case navigationManager.areaButtons:
        navigationManager.moveInButtonArea(direction)
        break
    }

    event.accepted = true
}
```

**关键功能**：
- 处理方向键（Up, Down, Left, Right）
- 处理回车键（触发当前焦点项）
- 根据当前区域调用对应的导航函数

#### 2.6 添加 Connections 对象（Line 1473-1512）

```qml
// ✅ 2026-01-31 [FIX 100.300.112.8]: TensionControlPage 焦点同步
Connections {
    target: root
    enabled: tensionControlPageLoader.item !== null

    function onCurrentFocusAreaChanged() {
        if (tensionControlPageLoader.item && root.currentCategory === 5) {
            if (root.currentFocusArea === 2) {
                // 焦点进入内容区域，默认在控制列表区域
                tensionControlPageLoader.item.focusSubArea = 0
                tensionControlPageLoader.item.focusItemIndex = root.currentContentItemIndex
            } else {
                // 焦点离开内容区域，清除焦点
                tensionControlPageLoader.item.focusItemIndex = -1
            }
        }
    }

    function onCurrentCategoryChanged() {
        if (tensionControlPageLoader.item) {
            if (root.currentCategory === 5 && root.currentFocusArea === 2) {
                // 切换到张紧控制类别，设置焦点
                tensionControlPageLoader.item.focusSubArea = 0
                tensionControlPageLoader.item.focusItemIndex = root.currentContentItemIndex
            } else {
                // 切换到其他类别，清除焦点
                tensionControlPageLoader.item.focusItemIndex = -1
            }
        }
    }

    function onCurrentContentItemIndexChanged() {
        if (tensionControlPageLoader.item &&
            root.currentFocusArea === 2 &&
            root.currentCategory === 5) {
            // 在控制列表中导航
            tensionControlPageLoader.item.focusItemIndex = root.currentContentItemIndex
        }
    }
}
```

**关键功能**：
- 监听 currentFocusArea 变化，同步焦点状态
- 监听 currentCategory 变化，处理类别切换
- 监听 currentContentItemIndex 变化，同步列表导航

## 🎯 导航架构

### 4 区域导航系统

```
TensionControlPage 导航架构

区域0: 控制列表（2个控制）
  ├─ 上键：上一个控制
  ├─ 下键：下一个控制
  └─ 右键：进入使用状态区域 ──────────┐
                                        │
区域1: 使用状态（2个选项）← ───────────┘
  ├─ 左键：返回控制列表
  ├─ 右键：进入参数区域
  ├─ 回车：切换投入/禁用
  └─ 下键：进入参数区域 ────────────┐
                                      │
区域2: 参数区域（N个参数）← ────────┘
  ├─ 左键（左列）：返回使用状态
  ├─ 右键（左列）：移动到右列
  ├─ 上键（第一行）：返回使用状态
  ├─ 下键（最后一行）：进入按钮区域 ──┐
  └─ 回车：打开虚拟键盘                │
                                        │
区域3: 按钮区域（3个按钮）← ───────────┘
  ├─ 左键（第一个）：返回使用状态
  ├─ 右键：移动到下一个按钮
  ├─ 上键：返回参数区域
  └─ 回车：触发按钮操作
```

### 焦点指示器规则

| 区域 | focusSubArea | 焦点属性 | 显示条件 |
|------|--------------|----------|----------|
| 控制列表 | 0 | focusItemIndex | `focusSubArea === 0 && focusItemIndex === index` |
| 使用状态 | 1 | focusUsageStatusIndex | `focusSubArea === 1 && focusUsageStatusIndex === index` |
| 参数区域 | 2 | focusParamIndex | `focusSubArea === 2 && focusParamIndex === index` |
| 按钮区域 | 3 | focusButtonIndex | `focusSubArea === 3 && focusButtonIndex === index` |

## 🔗 事件流

### 键盘事件处理流程

```
用户按键
  ↓
DeviceSettingsDialog.Keys.onXPressed 触发
  ↓
检查 currentCategory === 5 && currentFocusArea === 2
  ↓
调用 tensionPage.handleKeyPress(direction)
  ↓
TensionControlPage.handleKeyPress 被调用
  ↓
根据 navigationManager.currentArea 调用对应的导航函数
  ↓
更新焦点属性（focusSubArea, focusItemIndex, focusParamIndex, focusButtonIndex）
  ↓
UI 更新（焦点指示器移动）
```

### 焦点同步流程

```
DeviceSettingsDialog.currentFocusArea 变化
  ↓
Connections.onCurrentFocusAreaChanged 触发
  ↓
检查 currentCategory === 5
  ↓
如果 currentFocusArea === 2：
  设置 tensionPage.focusSubArea = 0
  设置 tensionPage.focusItemIndex = currentContentItemIndex
否则：
  清除 tensionPage.focusItemIndex = -1
```

## 💡 关键设计模式

### 1. NavigationManager 模式

**优点**：
- 集中管理导航逻辑
- 区域和索引分离
- 易于维护和扩展

**实现**：
```qml
DeviceInfo.NavigationManager {
    id: navigationManager

    // 区域定义
    readonly property int areaXXX: 0

    // 当前状态
    property int currentArea: areaXXX
    property int xxxIndex: 0

    // 导航函数
    function moveInXXXArea(direction) { ... }
    function switchToArea(newArea) { ... }
}
```

### 2. 事件委托模式

**优点**：
- DeviceSettingsDialog 只负责检测和转发
- 页面内部处理具体导航逻辑
- 解耦合，易于维护

**实现**：
```qml
// DeviceSettingsDialog
if (currentCategory === 5 && currentFocusArea === 2) {
    tensionPage.handleKeyPress(direction)
    return
}

// TensionControlPage
function handleKeyPress(direction) {
    navigationManager.moveInXXXArea(direction)
}
```

### 3. 属性绑定模式

**优点**：
- 自动同步，无需手动更新
- 响应式更新
- 避免绑定循环

**实现**：
```qml
// 使用 Qt.binding() 确保绑定不断开
item.focusItemIndex = Qt.binding(function() {
    return root.focusItemIndex
})
```

### 4. Connections 模式

**优点**：
- 避免动态绑定导致的事件循环阻塞
- 明确的信号处理
- 易于调试

**实现**：
```qml
Connections {
    target: root
    enabled: pageLoader.item !== null

    function onCurrentFocusAreaChanged() {
        // 处理焦点区域变化
    }
}
```
