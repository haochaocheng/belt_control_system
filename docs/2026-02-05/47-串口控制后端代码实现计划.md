## 🔌 Phase 2: 串口打开/关闭功能

### 2.1 打开串口

```cpp
bool SerialPortController::openSerialPort() {
    if (m_currentPortIndex < 0 || m_currentPortIndex >= m_serialPortConfigs.size()) {
        emit errorOccurred("无效的串口索引");
        return false;
    }

    SerialPortConfig &config = m_serialPortConfigs[m_currentPortIndex];

    // 如果已经打开，先关闭
    if (m_currentSerialPort && m_currentSerialPort->isOpen()) {
        m_currentSerialPort->close();
    }

    // 创建或获取串口对象
    m_currentSerialPort = getOrCreateSerialPort(m_currentPortIndex);

    // 设置串口参数
    m_currentSerialPort->setPortName(config.devicePath);
    m_currentSerialPort->setBaudRate(config.baudRate);
    m_currentSerialPort->setDataBits(static_cast<QSerialPort::DataBits>(config.dataBits));
    m_currentSerialPort->setStopBits(static_cast<QSerialPort::StopBits>(config.stopBits));

    // 设置校验位
    if (config.parity == "None") {
        m_currentSerialPort->setParity(QSerialPort::NoParity);
    } else if (config.parity == "Even") {
        m_currentSerialPort->setParity(QSerialPort::EvenParity);
    } else if (config.parity == "Odd") {
        m_currentSerialPort->setParity(QSerialPort::OddParity);
    }

    // 打开串口
    if (!m_currentSerialPort->open(QIODevice::ReadWrite)) {
        QString error = QString("无法打开串口 %1: %2")
                        .arg(config.portName)
                        .arg(m_currentSerialPort->errorString());
        emit errorOccurred(error);
        return false;
    }

    // 更新状态
    config.isOpen = true;
    emit isOpenChanged();

    qDebug() << "✅ [SerialPortController] 串口已打开:" << config.portName;
    return true;
}
```

### 2.2 关闭串口

```cpp
void SerialPortController::closeSerialPort() {
    if (!m_currentSerialPort || !m_currentSerialPort->isOpen()) {
        return;
    }

    m_currentSerialPort->close();

    // 更新状态
    if (m_currentPortIndex >= 0 && m_currentPortIndex < m_serialPortConfigs.size()) {
        m_serialPortConfigs[m_currentPortIndex].isOpen = false;
        emit isOpenChanged();
    }

    qDebug() << "✅ [SerialPortController] 串口已关闭";
}
```

## 📤 Phase 3: 数据发送功能

### 3.1 发送数据

```cpp
void SerialPortController::sendData(const QString &data, bool isHex) {
    if (!m_currentSerialPort || !m_currentSerialPort->isOpen()) {
        emit errorOccurred("串口未打开");
        return;
    }

    QByteArray sendData;

    if (isHex) {
        // HEX 模式：将十六进制字符串转换为字节数组
        sendData = hexStringToByteArray(data);
    } else {
        // ASCII 模式：直接转换为字节数组
        sendData = data.toUtf8();
    }

    qint64 bytesWritten = m_currentSerialPort->write(sendData);

    if (bytesWritten == -1) {
        emit errorOccurred("发送数据失败");
    } else {
        qDebug() << "✅ [SerialPortController] 发送数据:" << bytesWritten << "字节";
    }
}
```

### 3.2 HEX 字符串转换

```cpp
QByteArray SerialPortController::hexStringToByteArray(const QString &hex) {
    QByteArray result;
    QString cleanHex = hex.simplified().remove(' ');

    for (int i = 0; i < cleanHex.length(); i += 2) {
        QString byteString = cleanHex.mid(i, 2);
        bool ok;
        quint8 byte = byteString.toUInt(&ok, 16);
        if (ok) {
            result.append(byte);
        }
    }

    return result;
}
```

## 📥 Phase 4: 数据接收功能

### 4.1 接收数据处理

```cpp
void SerialPortController::handleReadyRead() {
    if (!m_currentSerialPort) {
        return;
    }

    QByteArray data = m_currentSerialPort->readAll();

    // 发送原始数据信号
    emit dataReceived(data);

    // 更新接收缓冲区
    if (m_receiveHexMode) {
        // HEX 模式：转换为十六进制字符串
        m_receiveBuffer += byteArrayToHexString(data) + " ";
    } else {
        // ASCII 模式：直接转换为字符串
        m_receiveBuffer += QString::fromUtf8(data);
    }

    emit receiveBufferChanged();

    // 如果是 MODBUS 响应，添加到 MODBUS 缓冲区
    if (m_modbusTimer->isActive()) {
        m_modbusBuffer.append(data);
        // 检查是否接收完整
        if (isModbusResponseComplete(m_modbusBuffer)) {
            m_modbusTimer->stop();
            parseAndEmitModbusResponse();
        }
    }
}
```

### 4.2 字节数组转HEX字符串

```cpp
QString SerialPortController::byteArrayToHexString(const QByteArray &data) {
    QString result;
    for (quint8 byte : data) {
        result += QString("%1 ").arg(byte, 2, 16, QChar('0')).toUpper();
    }
    return result.trimmed();
}
```

## 🔧 Phase 5: MODBUS RTU 协议实现

### 5.1 读保持寄存器 (功能码 0x03)

```cpp
void SerialPortController::readHoldingRegisters(int slaveAddress, int startAddress, int quantity) {
    if (!m_currentSerialPort || !m_currentSerialPort->isOpen()) {
        emit errorOccurred("串口未打开");
        return;
    }

    // 构建 MODBUS 请求
    QByteArray request = buildModbusRequest(slaveAddress, 0x03, startAddress, quantity);

    // 发送请求
    m_currentSerialPort->write(request);

    // 启动超时定时器
    m_pendingSlaveAddress = slaveAddress;
    m_pendingFunctionCode = 0x03;
    m_modbusBuffer.clear();
    m_modbusTimer->start(1000);  // 1秒超时

    qDebug() << "✅ [MODBUS] 读保持寄存器 - 从站:" << slaveAddress
             << "起始地址:" << startAddress << "数量:" << quantity;
}
```

### 5.2 写单个寄存器 (功能码 0x06)

```cpp
void SerialPortController::writeSingleRegister(int slaveAddress, int address, int value) {
    if (!m_currentSerialPort || !m_currentSerialPort->isOpen()) {
        emit errorOccurred("串口未打开");
        return;
    }

    // 构建 MODBUS 请求
    QList<int> values = {value};

    m_currentSerialPort->write(request);

    // 启动超时定时器
    m_pendingSlaveAddress = slaveAddress;
    m_pendingFunctionCode = 0x06;
    m_modbusBuffer.clear();
    m_modbusTimer->start(1000);

             << "地址:" << address << "值:" << value;
}
```

### 5.3 构建 MODBUS 请求

```cpp
QByteArray SerialPortController::buildModbusRequest(int slaveAddress, int functionCode,
                                                     int startAddress, int quantity,
                                                     const QList<int> &values) {
    QByteArray request;

    // 从站地址
    request.append(static_cast<char>(slaveAddress));

    // 功能码
    request.append(static_cast<char>(functionCode));

    // 起始地址（高字节在前）
    request.append(static_cast<char>((startAddress >> 8) & 0xFF));
    request.append(static_cast<char>(startAddress & 0xFF));

    if (functionCode == 0x03) {
        // 读保持寄存器：数量（高字节在前）
        request.append(static_cast<char>((quantity >> 8) & 0xFF));
        request.append(static_cast<char>(quantity & 0xFF));
    } else if (functionCode == 0x06) {
        // 写单个寄存器：值（高字节在前）
        int value = values.isEmpty() ? 0 : values.first();
        request.append(static_cast<char>((value >> 8) & 0xFF));
        request.append(static_cast<char>(value & 0xFF));
    } else if (functionCode == 0x10) {
        // 写多个寄存器：数量 + 字节数 + 值
        request.append(static_cast<char>((quantity >> 8) & 0xFF));
        request.append(static_cast<char>(quantity & 0xFF));
        request.append(static_cast<char>(quantity * 2));  // 字节数

        for (int value : values) {
            request.append(static_cast<char>((value >> 8) & 0xFF));
            request.append(static_cast<char>(value & 0xFF));
        }
    }

    // 计算并添加 CRC16
    quint16 crc = calculateCRC16(request);
    request.append(static_cast<char>(crc & 0xFF));        // CRC 低字节
    request.append(static_cast<char>((crc >> 8) & 0xFF)); // CRC 高字节

    return request;
}
```

### 5.4 CRC16 计算

```cpp
quint16 SerialPortController::calculateCRC16(const QByteArray &data) {
    quint16 crc = 0xFFFF;

    for (quint8 byte : data) {
        crc ^= byte;
        for (int i = 0; i < 8; i++) {
            if (crc & 0x0001) {
                crc = (crc >> 1) ^ 0xA001;
            } else {
                crc >>= 1;
            }
        }
    }

    return crc;
}
```

### 6.1 类设计

**文件路径**: `src/control/ModbusRegisterModel.h`

```cpp
#ifndef MODBUSREGISTERMODEL_H
#define MODBUSREGISTERMODEL_H

#include <QAbstractTableModel>
#include <QMap>

struct ModbusRegister {
    int address;
    quint16 value;
    QString description;
};

class ModbusRegisterModel : public QAbstractTableModel
{
    Q_OBJECT

public:
    enum Roles {
        AddressRole = Qt::UserRole + 1,
        ValueHexRole,
        ValueDecRole,
        DescriptionRole
    };

    explicit ModbusRegisterModel(QObject *parent = nullptr);

    // QAbstractTableModel 接口
    int rowCount(const QModelIndex &parent = QModelIndex()) const override;
    int columnCount(const QModelIndex &parent = QModelIndex()) const override;
    QVariant data(const QModelIndex &index, int role = Qt::DisplayRole) const override;
    QVariant headerData(int section, Qt::Orientation orientation, int role = Qt::DisplayRole) const override;
    QHash<int, QByteArray> roleNames() const override;

    // 数据更新接口
    Q_INVOKABLE void updateRegister(int address, quint16 value);
    Q_INVOKABLE void updateRegisters(int startAddress, const QList<quint16> &values);
    Q_INVOKABLE void clearRegisters();

private:
    QMap<int, ModbusRegister> m_registers;
};

#endif // MODBUSREGISTERMODEL_H
```

### 6.2 实现要点

```cpp
void ModbusRegisterModel::updateRegisters(int startAddress, const QList<quint16> &values) {
    for (int i = 0; i < values.size(); i++) {
        int address = startAddress + i;
        quint16 value = values[i];

        if (m_registers.contains(address)) {
            // 更新现有寄存器
            m_registers[address].value = value;
        } else {
            // 添加新寄存器
            ModbusRegister reg;
            reg.address = address;
            reg.value = value;
            reg.description = QString("寄存器 %1").arg(address);
            m_registers[address] = reg;
        }
    }

    // 通知视图更新
    emit dataChanged(index(0, 0), index(rowCount() - 1, columnCount() - 1));
}
```

```cpp
void SerialPortController::saveConfig() {
    m_settings->beginGroup("SerialPorts");

    for (int i = 0; i < m_serialPortConfigs.size(); i++) {
        const SerialPortConfig &config = m_serialPortConfigs[i];

        m_settings->beginGroup(config.portName);
        m_settings->setValue("baudRate", config.baudRate);
        m_settings->setValue("dataBits", config.dataBits);
        m_settings->setValue("stopBits", config.stopBits);
        m_settings->setValue("parity", config.parity);
        m_settings->endGroup();
    }

    m_settings->endGroup();
    m_settings->sync();

    qDebug() << "✅ [SerialPortController] 配置已保存";
}
```

```cpp
void SerialPortController::loadConfig() {
    m_settings->beginGroup("SerialPorts");

    for (int i = 0; i < m_serialPortConfigs.size(); i++) {
        SerialPortConfig &config = m_serialPortConfigs[i];

        m_settings->beginGroup(config.portName);
        config.baudRate = m_settings->value("baudRate", 9600).toInt();

        config.stopBits = m_settings->value("stopBits", 1).toInt();
        config.parity = m_settings->value("parity", "None").toString();
        m_settings->endGroup();
    }

    m_settings->endGroup();

    qDebug() << "✅ [SerialPortController] 配置已加载";
}
```

## ⚠️ 注意事项

### 1. 串口权限

在 Linux 系统上，需要将用户添加到 `dialout` 组：

```bash
sudo usermod -a -G dialout $USER
```

在 Docker 容器中，需要挂载 `/dev` 目录：

```yaml
devices:
  - /dev:/dev
```

### 2. MODBUS RTU 协议

- **CRC16 校验**：使用标准 MODBUS CRC16 算法（多项式 0xA001）
- **超时处理**：默认 1 秒超时
- **错误码**：解析 MODBUS 错误码并显示友好提示

### 3. 性能优化

- **接收缓冲区限制**：最多保留 10000 行，超过则清除旧数据
- **定时器优化**：避免频繁更新 UI
- **内存管理**：及时清理不用的数据

### 4. 线程安全

- QSerialPort 在主线程中使用
- 避免在信号处理函数中进行耗时操作
- 使用 Qt::QueuedConnection 连接信号
