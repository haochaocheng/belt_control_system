### 关键功能特性

1. **GridLayout 4列布局**：
   - 列定义：标签1（160px）、输入框1（fillWidth, max 300）、标签2（160px）、输入框2（fillWidth, max 300）
   - 列间距：10px
   - 行间距：12px
   - 总宽度：`paramScrollView.width * 0.9`

2. **CustomSpinBox 虚拟键盘支持**：
   - 内置 `activateVirtualKeyboard()` 方法
   - 自动滚动到可见位置
   - 支持数字输入

3. **CustomComboBox 下拉选择**：
   - 回车键切换选项：`currentIndex = (currentIndex + 1) % model.length`
   - 不需要虚拟键盘

4. **焦点管理**：
   - 焦点指示器：蓝色边框（#2196F3），宽度3px，z值1000
   - 鼠标点击发射 `requestFocusParamIndex(paramIndex)` 信号
   - 焦点索引通过 `focusParamIndex` 属性传递

5. **条件显示/启用**：
   - 次数设置：`enabled: alarmTypeField.currentIndex === 0`
   - 时间设置：`enabled: alarmTypeField.currentIndex === 1`
   - 标签颜色：启用时 `#9E9E9E`，禁用时 `#3E3E3E`

**示例代码**（是否投入）：
```qml

Item {
    Layout.column: 1
    Layout.row: 0
    Layout.fillWidth: true
    Layout.maximumWidth: 300
    implicitHeight: enabledField.implicitHeight

    DeviceInfo.CustomComboBox {
        id: enabledField
        anchors.fill: parent
        model: ["投入", "禁用"]
        currentIndex: 0  // 默认选中"投入"
        keyboardManager: root.keyboardManager
    }

    // 鼠标点击发射信号
    MouseArea {
        anchors.fill: parent
        onClicked: function(mouse) {
            console.log("✅ [FrontBearingTempTab] 鼠标点击是否投入，发射信号: requestFocusParamIndex(0)")
            root.requestFocusParamIndex(0)
            mouse.accepted = false
        }
    }

    // 焦点指示器
    Rectangle {
        anchors.fill: parent
        color: "transparent"
        border.color: (root.focusParamIndex === 0) ? "#2196F3" : "transparent"
        border.width: (root.focusParamIndex === 0) ? 3 : 0
        radius: 4
        z: 1000
        enabled: false
    }
}
```

```qml
function triggerParamInput(paramIndex) {
    console.log("✅ [FrontBearingTempTab] 触发参数输入 - 索引:", paramIndex)

    var inputField = null

    switch(paramIndex) {
    case 0:  // 是否投入（CustomComboBox）
        console.log("✅ [FrontBearingTempTab] 切换是否投入")
        enabledField.currentIndex = (enabledField.currentIndex + 1) % enabledField.model.length
        break
    case 1:  // 报警类型（CustomComboBox）
        console.log("✅ [FrontBearingTempTab] 切换报警类型")
        alarmTypeField.currentIndex = (alarmTypeField.currentIndex + 1) % alarmTypeField.model.length
        break
    case 2:  // 次数设置（CustomSpinBox，条件启用）
        console.log("✅ [FrontBearingTempTab] 次数设置")
        if (alarmTypeField.currentIndex === 0) {
            inputField = countSettingField
        }
        break
    case 3:  // 时间设置（CustomSpinBox，条件启用）
        console.log("✅ [FrontBearingTempTab] 时间设置")
        if (alarmTypeField.currentIndex === 1) {
            inputField = timeSettingField
        }
        break
    case 4:  // 动作保护类型（CustomComboBox）
        console.log("✅ [FrontBearingTempTab] 动作保护类型")
        actionProtectionField.currentIndex = (actionProtectionField.currentIndex + 1) % actionProtectionField.model.length
        break
    case 5:  // 故障保护类型（CustomComboBox）
        console.log("✅ [FrontBearingTempTab] 故障保护类型")
        faultProtectionField.currentIndex = (faultProtectionField.currentIndex + 1) % faultProtectionField.model.length
        break
    case 6:  // 温度量程（CustomSpinBox）
        console.log("✅ [FrontBearingTempTab] 温度量程")
        inputField = tempRangeField
        break
    case 7:  // 温度上限（CustomSpinBox）
        console.log("✅ [FrontBearingTempTab] 温度上限")
        inputField = tempUpperField
        break
    case 8:  // 温度下限（CustomSpinBox）
        console.log("✅ [FrontBearingTempTab] 温度下限")
        inputField = tempLowerField
        break
    case 9:  // 输入点选择（CustomComboBox）
        console.log("✅ [FrontBearingTempTab] 输入点选择")
        inputPointField.currentIndex = (inputPointField.currentIndex + 1) % inputPointField.model.length
        break
    case 10:  // 过滤干扰延时（CustomSpinBox）
        console.log("✅ [FrontBearingTempTab] 过滤干扰延时")
        inputField = filterDelayField
        break
    }

    // 激活虚拟键盘
    if (inputField) {
        console.log("✅ [FrontBearingTempTab] 激活虚拟键盘 - 控件:", inputField)
        if (inputField.activateVirtualKeyboard) {
            inputField.activateVirtualKeyboard()
        } else {
            inputField.forceActiveFocus()
        }
    }
}
```
