**格式**: `{引擎名}-{模型名}-{说话人ID}/`

| 引擎 | 模型 | 说话人ID | 文件夹名 |
|------|------|---------|---------|
| PaddleSpeech | FastSpeech2-aishell3 | 174 | `paddlespeech-fastspeech2-aishell3-spk174` |
| PaddleSpeech | FastSpeech2-aishell3 | 21 | `paddlespeech-fastspeech2-aishell3-spk21` |

| MeloTTS | Chinese | 0 | `melotts-chinese-spk0` |

### 3. 文件命名规则

#### 3.1 开关量保护（8种）
```
{设备号}号皮带{保护名称}保护.wav
```
示例：`1号皮带沿线急停保护.wav`

#### 3.2 模拟量保护（8种状态）
```
{设备号}号皮带{保护名称}保护.wav
```
示例：`1号皮带速度超速保护.wav`、`1号皮带低速打滑保护.wav`

#### 3.3 电机保护（9种 × 4台电机）
```
{设备号}号皮带{电机号}号电机{保护名称}保护.wav
```
示例：`1号皮带1号电机电流过载保护.wav`

#### 3.4 制动器保护（3种 × 4台制动器）
```
{设备号}号皮带{制动器号}号制动器{保护名称}保护.wav
```
示例：`1号皮带1号制动器温度过高保护.wav`

#### 3.5 张紧保护（3种 × 2套张紧）
```
{设备号}号皮带{张紧号}号张紧装置{保护名称}保护.wav
```
示例：`1号皮带1号张紧装置张紧力过大保护.wav`

#### 3.6 沿线点位保护（3种 × 64点位）
```
{设备号}号皮带{点位号}号沿线{保护名称}保护.wav
```
示例：`1号皮带32号沿线急停保护.wav`

## 🔧 类设计

### AudioPathMapper 类

```cpp
// src/control/AudioPathMapper.h
#ifndef AUDIOPATHMAPPER_H
#define AUDIOPATHMAPPER_H

#include <QString>

class AudioPathMapper
{
public:
    // 保护类型枚举
    enum ProtectionType {
        SwitchProtection,    // 开关量保护
        AnalogProtection,    // 模拟量保护
        MotorProtection,     // 电机保护
        BrakeProtection,     // 制动器保护
        TensionProtection,   // 张紧保护
        LineProtection       // 沿线点位保护
    };

    // 获取音频文件路径
    static QString getAudioPath(
        int deviceId,                    // 设备号 1-8
        ProtectionType type,             // 保护类型
        const QString &protectionName,   // 保护名称
        int subIndex = 0                 // 子索引（电机号/点位号）
    );

    // 获取 TTS 文本
    static QString getTtsText(
        int deviceId,
        ProtectionType type,
        const QString &protectionName,
        int subIndex = 0
    );

    // 检查音频文件是否存在
    static bool audioFileExists(const QString &path);

    // 设置音频基础路径
    static void setAudioBasePath(const QString &path);

    // 设置当前使用的引擎目录
    static void setEngineDir(const QString &engineDir);

private:
    static QString m_audioBasePath;  // 如 /app/AUDIO
    static QString m_engineDir;      // 如 paddlespeech-fastspeech2-aishell3-spk21
};

#endif // AUDIOPATHMAPPER_H
```

### 实现核心逻辑

```cpp
QString AudioPathMapper::m_audioBasePath = "/app/AUDIO";
QString AudioPathMapper::m_engineDir = "default";

QString AudioPathMapper::getAudioPath(
    int deviceId,
    ProtectionType type,
    const QString &protectionName,
    int subIndex)
{
    // 皮带目录
    QString beltDir = QString("%1#PD").arg(deviceId);
    QString fileName;

    switch (type) {
    case SwitchProtection:
        // 开关量保护：1号皮带沿线急停保护.wav
        fileName = QString("%1号皮带%2保护.wav")
                    .arg(deviceId).arg(protectionName);
        break;

    case AnalogProtection:
        // 模拟量保护：1号皮带速度超速保护.wav
        fileName = QString("%1号皮带%2保护.wav")
                    .arg(deviceId).arg(protectionName);
        break;

    case MotorProtection:
        // 电机保护：1号皮带1号电机电流过载保护.wav
        fileName = QString("%1号皮带%2号电机%3保护.wav")
                    .arg(deviceId).arg(subIndex).arg(protectionName);
        break;

    case BrakeProtection:
        // 制动器保护：1号皮带1号制动器温度过高保护.wav
        fileName = QString("%1号皮带%2号制动器%3保护.wav")
                    .arg(deviceId).arg(subIndex).arg(protectionName);
        break;

    case TensionProtection:
        // 张紧保护：1号皮带1号张紧装置张紧力过大保护.wav
        fileName = QString("%1号皮带%2号张紧装置%3保护.wav")
                    .arg(deviceId).arg(subIndex).arg(protectionName);
        break;

    case LineProtection:
        // 沿线点位保护：1号皮带32号沿线急停保护.wav
        fileName = QString("%1号皮带%2号沿线%3保护.wav")
                    .arg(deviceId).arg(subIndex).arg(protectionName);
        break;
    }

    // 完整路径：/app/AUDIO/default/1#PD/1号皮带沿线急停保护.wav
    return QString("%1/%2/%3/%4")
            .arg(m_audioBasePath)
            .arg(m_engineDir)
            .arg(beltDir)
            .arg(fileName);
}
```

## 🔄 工作流程

```
保护触发
    ↓
ProtectionMonitorService.triggerProtection()
    ↓
AudioPathMapper.getAudioPath() → 获取音频路径
AudioPathMapper.getTtsText()   → 获取TTS文本
    ↓
emit alarmTriggered(protectionName, ttsText, audioFile, ...)
    ↓
AlarmPlaybackService.onAlarmTriggered()
    ↓
文件存在? → 是 → playAudioFile()
    ↓ 否
synthesizeAndCache() → 实时合成并保存到相同路径
    ↓
playAudioFile()
```
