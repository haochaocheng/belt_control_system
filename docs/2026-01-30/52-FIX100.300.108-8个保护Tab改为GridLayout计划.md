## 🔧 修改要点

### 1. 布局模式属性

```qml
// 修改前
readonly property string layoutMode: "two-column"

// 修改后
readonly property string layoutMode: "grid"
```

### 2. ScrollView ID

```qml
// 修改前
ScrollView {
    anchors.fill: parent
    clip: true

// 修改后
ScrollView {
    id: paramScrollView  // ✅ 添加 ID
    anchors.fill: parent
    clip: true
```

### 3. GridLayout 配置

```qml
GridLayout {
    width: paramScrollView.width * 0.9  // ✅ 占 ScrollView 宽度的 90%
    columns: 4  // 4列：标签1、输入框1、标签2、输入框2
    columnSpacing: 10
    rowSpacing: 12
```

### 4. 标签和输入框定位

使用 `Layout.column` 和 `Layout.row` 精确定位：

```qml
// 第一行左侧（索引0）
Text {
    text: "是否投入:"
    Layout.column: 0
    Layout.row: 0
    Layout.preferredWidth: 160
    ...
}

Item {
    Layout.column: 1
    Layout.row: 0
    Layout.fillWidth: true
    Layout.maximumWidth: 300
    implicitHeight: enabledRow.implicitHeight
    ...
}

// 第一行右侧（索引1）
Text {
    text: "报警类型:"
    Layout.column: 2
    Layout.row: 0
    Layout.preferredWidth: 160
    ...
}

Item {
    Layout.column: 3
    Layout.row: 0
    Layout.fillWidth: true
    Layout.maximumWidth: 300
    implicitHeight: alarmTypeRow.implicitHeight
    ...
}
```

### 5. 动态高度计算

```qml
// RadioButton 组
Item {
    implicitHeight: enabledRow.implicitHeight  // ✅ 引用 Row 的 implicitHeight

    Row {
        id: enabledRow
        anchors.fill: parent
        ...
    }
}

// CustomReadOnlyField
Item {
    implicitHeight: actionProtectionField.implicitHeight  // ✅ 引用 CustomReadOnlyField 的 implicitHeight

    DeviceInfo.CustomReadOnlyField {
        id: actionProtectionField
        anchors.fill: parent
        ...
    }
}

// Row 容器（带单位）
Item {
    implicitHeight: currentUpperRow.implicitHeight  // ✅ 引用 Row 的 implicitHeight

    Row {
        id: currentUpperRow
        width: parent.width
        height: 60  // ✅ 设置 Row 的固定高度
        ...
    }
}
```
