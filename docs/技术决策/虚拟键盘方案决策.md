## 🔧 技术实现

### 当前实现

**集成方式**：
```qml
// QtVirtualKeyboardIntegration.qml
import QtQuick.VirtualKeyboard 6.5

Popup {
    id: root
    width: parent.width
    height: 300
    y: parent.height - height

    property var targetTextField: null
    property string inputMode: "numeric"

    function openForField(textField, callback, mode) {
        targetTextField = textField
        inputMode = mode || "numeric"

        // Set input method hints
        if (mode === "numeric") {
            textField.inputMethodHints = Qt.ImhDigitsOnly
        } else if (mode === "english") {
            textField.inputMethodHints = Qt.ImhNoPredictiveText | Qt.ImhPreferLowercase
        } else if (mode === "chinese") {
            textField.inputMethodHints = Qt.ImhNone
        }

        textField.forceActiveFocus()
        root.open()
    }

    contentItem: InputPanel {
        id: inputPanel
        anchors.fill: parent
        style: KeyboardStyle { }
    }
}
```

**使用方式**：
```qml
// DeviceSettingsDialog.qml
import "../../components/virtual_keyboard"

QtVirtualKeyboardIntegration {
    id: virtualKeyboard
}

// 打开虚拟键盘
virtualKeyboard.openForField(textField, callback, "numeric")
```
